function decode(el){
  if(RM||el.dataset.done)return;el.dataset.done='1';
  var txt=el.textContent,n=txt.length,f=0,steps=18;
  var iv=setInterval(function(){f++;var keep=Math.floor(n*f/steps),out=txt.slice(0,keep);
    for(var i=keep;i<n;i++)out+=txt[i]===' '?' ':(Math.random()<.5?'0':'1');
    el.textContent=out;if(f>=steps){clearInterval(iv);el.textContent=txt;}},38);
}
